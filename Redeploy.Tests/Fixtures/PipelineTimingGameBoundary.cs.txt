public static class Spy
{
    public static int Reads, Reflections, Scheduled, Infos, Warnings;
    public static bool ThrowOnClock;
    public static readonly List<string> Lines = new List<string>();
    public static void Reset() { Reads = Reflections = Scheduled = Infos = Warnings = 0; Lines.Clear(); ThrowOnClock = false; }
    public static void Require(bool ok, string why) { if (!ok) throw new Exception(why); }
}
public static class TimingProbe
{
    public static void Run()
    {
        RespawnPipelineTiming.ResetForRaidBoundary();
        Spy.Reset();
        var game = new EFT.LocalGame();
        var request = new RespawnRequest();
        Spy.Require(!request.EnablePipelineTiming, "new requests must default off");
        var off = RespawnPipelineTiming.Begin(game, request);
        off.Mark("create");
        var stamp = RespawnPipelineTiming.StartVictimsRestore();
        Spy.Require(stamp == 0, "disabled restore must not start a clock");
        RespawnPipelineTiming.RecordVictimsRestore(stamp, new EFT.Profile());
        off.Finish("replaced", "swap", game);
        Spy.Require(Spy.Reads == 0 && Spy.Reflections == 0 && Spy.Scheduled == 0 && Spy.Infos == 0 && Spy.Warnings == 0,
            "disabled collector touched external state, reflected, logged or scheduled work");

        request.EnablePipelineTiming = true;
        var on = RespawnPipelineTiming.Begin(game, request);
        Spy.Require(Spy.Reads > 0 && Spy.Reflections > 0, "enabled state probe did not run");
        stamp = RespawnPipelineTiming.StartVictimsRestore();
        Spy.Require(stamp != 0, "enabled restore clock did not start");
        RespawnPipelineTiming.RecordVictimsRestore(stamp, new EFT.Profile());
        on.Mark("create");
        on.Finish("replaced", "death", game);
        Spy.Require(Spy.Scheduled == 1 && Spy.Infos == 2 && Spy.Warnings == 0, "opt-in must collect both lines and exactly one coroutine");
        Spy.Require(Spy.Lines[0].Contains("Respawn pipeline #1") && Spy.Lines[0].Contains("3 victims"), "enabled state/restore details absent");
        on.Finish("replaced", "death", game);
        Spy.Require(Spy.Scheduled == 1 && Spy.Infos == 2, "Finish must be idempotent");

        Spy.Reset();
        request.EnablePipelineTiming = false;
        off = RespawnPipelineTiming.Begin(game, request);
        off.Mark("rewire");
        RespawnPipelineTiming.RecordVictimsRestore(1, new EFT.Profile());
        off.Finish("replaced", "death", game);
        Spy.Require(RespawnPipelineTiming.StartVictimsRestore() == 0 && Spy.Reads == 0 && Spy.Infos == 0 && Spy.Scheduled == 0,
            "later disabled request inherited previous collection");

        Spy.Reset();
        request.EnablePipelineTiming = true;
        on = RespawnPipelineTiming.Begin(game, request);
        RespawnPipelineTiming.ResetForRaidBoundary();
        Spy.Require(RespawnPipelineTiming.StartVictimsRestore() == 0, "raid boundary retained active restore timing");
        Spy.Reset();
        Spy.ThrowOnClock = true;
        var failed = RespawnPipelineTiming.Begin(game, request);
        failed.Mark("failed");
        failed.Finish("replaced", "swap", game);
        Spy.Require(Spy.Warnings == 1 && Spy.Infos == 0 && Spy.Scheduled == 0
            && RespawnPipelineTiming.StartVictimsRestore() == 0, "probe failure escaped or left collection active");
    }
}
